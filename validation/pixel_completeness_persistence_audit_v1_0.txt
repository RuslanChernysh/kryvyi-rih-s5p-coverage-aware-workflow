// ARTICLE4_PIXEL_COMPLETENESS_PERSISTENCE_AUDIT_V1_0
// Read-only diagnostic audit for Kryvyi Rih, 2019-2024.
// Pollutants: NO2 and CO. No exports or source assets are modified.

var AUDIT = {
  startYear: 2019, endYear: 2024,
  aoiAsset: 'projects/ee-ruslan777chernysh/assets/KRYVYI_RIH_AOI',
  reductionCrs: 'EPSG:4326', scaleM: 1113.2,
  tileScale: 8, maxPixels: 1e9, hotspotPercentile: 90
};

var POLLUTANTS = {
  NO2: {
    collection: 'COPERNICUS/S5P/OFFL/L3_NO2',
    band: 'tropospheric_NO2_column_number_density',
    nominalOnly: true
  },
  CO: {
    collection: 'COPERNICUS/S5P/OFFL/L3_CO',
    band: 'CO_column_number_density',
    nominalOnly: false
  }
};

var AOI_FC = ee.FeatureCollection(AUDIT.aoiAsset);
var AOI = AOI_FC.geometry().dissolve(1);
var AOI_MASK = ee.Image.constant(1).rename('aoi').clip(AOI);

Map.centerObject(AOI_FC, 9);
Map.addLayer(AOI, {color: 'FF0000'}, 'AUDIT AOI', false);

print('==================================================');
print('PIXEL COMPLETENESS + PERSISTENCE AUDIT');
print('AOI asset:', AUDIT.aoiAsset);
print('AOI source feature count:', AOI_FC.size());
print('AOI area, km2:', AOI.area(1).divide(1e6));
print('Study period:', AUDIT.startYear + '-' + AUDIT.endYear);
print('Audit scale, m:', AUDIT.scaleM);
print('Audit CRS:', AUDIT.reductionCrs);
print('Hotspot percentile:', AUDIT.hotspotPercentile);

function emptyValueImage() {
  return ee.Image.constant(0).rename('VALUE').toFloat()
    .updateMask(ee.Image.constant(0)).clip(AOI);
}

function createSourceCollection(pollutantKey, startDate, endDate) {
  var cfg = POLLUTANTS[pollutantKey];
  var collection = ee.ImageCollection(cfg.collection)
    .filterBounds(AOI).filterDate(startDate, endDate);

  if (cfg.nominalOnly) {
    collection = collection.filter(ee.Filter.eq('PRODUCT_QUALITY', 'NOMINAL'));
  }

  return collection.select(cfg.band).map(function(image) {
    return ee.Image(image).rename('VALUE').toFloat()
      .copyProperties(image, ['system:time_start']);
  });
}

function createDailyFirstMonthlyComposite(pollutantKey, year, month) {
  var start = ee.Date.fromYMD(year, month, 1);
  var end = start.advance(1, 'month');
  var monthlySource = createSourceCollection(pollutantKey, start, end);
  var numberOfDays = end.difference(start, 'day');
  var dayOffsets = ee.List.sequence(0, numberOfDays.subtract(1));

  var dailyImages = ee.ImageCollection.fromImages(dayOffsets.map(function(offset) {
    offset = ee.Number(offset);
    var dayStart = start.advance(offset, 'day');
    var dayEnd = dayStart.advance(1, 'day');
    var dayCollection = monthlySource.filterDate(dayStart, dayEnd);

    var dayMean = ee.Image(ee.Algorithms.If(
      dayCollection.size().gt(0),
      dayCollection.mean().rename('VALUE').toFloat(),
      emptyValueImage()
    ));

    return dayMean.set('system:time_start', dayStart.millis());
  }));

  return dailyImages.mean().rename('VALUE').toFloat().clip(AOI).set({
    pollutant: pollutantKey, year: year, month: month,
    'system:time_start': start.millis()
  });
}

function monthlyValidityImage(monthlyImage) {
  return ee.Image(monthlyImage).mask().select(0).gt(0).unmask(0)
    .rename('valid_month').toByte().updateMask(AOI_MASK);
}

function countValidPixels(binaryImage, outputBandName) {
  return ee.Image(binaryImage).rename(outputBandName).selfMask().reduceRegion({
    reducer: ee.Reducer.count(), geometry: AOI,
    crs: AUDIT.reductionCrs, scale: AUDIT.scaleM,
    maxPixels: AUDIT.maxPixels, tileScale: AUDIT.tileScale
  }).get(outputBandName);
}

function histogramImage(image, bandName) {
  return ee.Image(image).rename(bandName).reduceRegion({
    reducer: ee.Reducer.frequencyHistogram(), geometry: AOI,
    crs: AUDIT.reductionCrs, scale: AUDIT.scaleM,
    maxPixels: AUDIT.maxPixels, tileScale: AUDIT.tileScale
  }).get(bandName);
}

function buildAnnualAudit(pollutantKey, year) {
  var monthlyImages = [];
  for (var month = 1; month <= 12; month++) {
    monthlyImages.push(createDailyFirstMonthlyComposite(pollutantKey, year, month));
  }

  var monthlyCollection = ee.ImageCollection.fromImages(monthlyImages);
  var validMonthImages = monthlyImages.map(function(image) {
    return monthlyValidityImage(image);
  });

  var validMonthCount = ee.ImageCollection.fromImages(validMonthImages).sum()
    .rename('valid_month_count').toByte().updateMask(AOI_MASK);

  var strict12Mask = validMonthCount.eq(12).rename('strict_12_of_12')
    .toByte().updateMask(AOI_MASK);

  var annualMean = monthlyCollection.mean().rename('VALUE').toFloat().clip(AOI);

  var annualValidMask = annualMean.mask().select(0).gt(0).unmask(0)
    .rename('annual_valid').toByte().updateMask(AOI_MASK);

  // Robust P90 extraction: use the actual dictionary key returned by GEE.
  var p90Dictionary = annualMean.reduceRegion({
    reducer: ee.Reducer.percentile([AUDIT.hotspotPercentile]),
    geometry: AOI, crs: AUDIT.reductionCrs, scale: AUDIT.scaleM,
    maxPixels: AUDIT.maxPixels, tileScale: AUDIT.tileScale
  });

  var p90Keys = p90Dictionary.keys();
  print(pollutantKey + ' ' + year + ' P90 dictionary:', p90Dictionary);
  print(pollutantKey + ' ' + year + ' P90 dictionary keys:', p90Keys);

  var p90Threshold = ee.Number(p90Dictionary.get(p90Keys.get(0)));

  var hotspot = annualMean.gte(p90Threshold).rename('hotspot').toByte()
    .updateMask(annualMean.mask());

  return {
    validMonthCount: validMonthCount,
    strict12Mask: strict12Mask,
    annualMean: annualMean,
    annualValidMask: annualValidMask,
    p90Threshold: p90Threshold,
    hotspot: hotspot
  };
}

function runPollutantAudit(pollutantKey) {
  print('==================================================');
  print('POLLUTANT:', pollutantKey);

  var strictMasks = [], annualValidMasks = [], hotspotImages = [];

  for (var year = AUDIT.startYear; year <= AUDIT.endYear; year++) {
    var result = buildAnnualAudit(pollutantKey, year);

    strictMasks.push(result.strict12Mask.unmask(0).updateMask(AOI_MASK));
    annualValidMasks.push(result.annualValidMask.unmask(0).updateMask(AOI_MASK));
    hotspotImages.push(result.hotspot);

    print('----------------------------------------------');
    print(pollutantKey + ' ' + year + ' valid-month histogram:',
      histogramImage(result.validMonthCount, 'valid_month_count'));
    print(pollutantKey + ' ' + year + ' cells with 12/12 valid months:',
      countValidPixels(result.strict12Mask, 'strict_12'));
    print(pollutantKey + ' ' + year + ' cells with any annual result:',
      countValidPixels(result.annualValidMask, 'annual_valid'));
    print(pollutantKey + ' ' + year + ' P90 threshold:', result.p90Threshold);
    print(pollutantKey + ' ' + year + ' hotspot cells:',
      countValidPixels(result.hotspot, 'hotspot'));
  }

  var commonStrictMask = ee.ImageCollection.fromImages(strictMasks).min().eq(1)
    .rename('common_strict_6yr').toByte().updateMask(AOI_MASK);

  var commonAnnualValidMask = ee.ImageCollection.fromImages(annualValidMasks).min().eq(1)
    .rename('common_annual_valid_6yr').toByte().updateMask(AOI_MASK);

  var unionAnnualValidMask = ee.ImageCollection.fromImages(annualValidMasks).max().eq(1)
    .rename('union_annual_valid_6yr').toByte().updateMask(AOI_MASK);

  var intermittentAnnualMask = unionAnnualValidMask.and(commonAnnualValidMask.not())
    .rename('intermittent_annual_validity').toByte().updateMask(AOI_MASK);

  var originalLikePersistence = ee.ImageCollection.fromImages(
    hotspotImages.map(function(image) {
      return ee.Image(image).unmask(0).rename('hotspot').toByte().updateMask(AOI_MASK);
    })
  ).sum().rename('persistence_original_like').toByte().updateMask(AOI_MASK);

  var strictPersistence = ee.ImageCollection.fromImages(
    hotspotImages.map(function(image) {
      return ee.Image(image).unmask(0).rename('hotspot').toByte()
        .updateMask(commonStrictMask);
    })
  ).sum().rename('persistence_strict').toByte().updateMask(commonStrictMask);

  var originalSixOfSix = originalLikePersistence.eq(6)
    .rename('original_6_of_6').toByte();
  var strictSixOfSix = strictPersistence.eq(6)
    .rename('strict_6_of_6').toByte();

  var originalSixExcludedByStrict = originalSixOfSix
    .and(commonStrictMask.unmask(0).not())
    .rename('original_6_of_6_excluded_by_strict')
    .toByte().updateMask(AOI_MASK);

  print('==============================================');
  print(pollutantKey + ' COMMON 6-YEAR SUMMARY');
  print(pollutantKey + ' cells with annual result in ALL 6 years:',
    countValidPixels(commonAnnualValidMask, 'common_annual'));
  print(pollutantKey + ' cells valid in >=1 year but missing annual result in >=1 other year:',
    countValidPixels(intermittentAnnualMask, 'intermittent'));
  print(pollutantKey + ' cells with 12/12 valid months in ALL 6 years:',
    countValidPixels(commonStrictMask, 'common_strict'));
  print(pollutantKey + ' ORIGINAL-LIKE persistence histogram:',
    histogramImage(originalLikePersistence, 'persistence_original_like'));
  print(pollutantKey + ' STRICT persistence histogram:',
    histogramImage(strictPersistence, 'persistence_strict'));
  print(pollutantKey + ' ORIGINAL-LIKE 6/6 hotspot cells:',
    countValidPixels(originalSixOfSix, 'original_6'));
  print(pollutantKey + ' STRICT 6/6 hotspot cells:',
    countValidPixels(strictSixOfSix, 'strict_6'));
  print(pollutantKey + ' original 6/6 cells excluded by strict 12/12 common mask:',
    countValidPixels(originalSixExcludedByStrict, 'excluded'));

  Map.addLayer(commonStrictMask.selfMask(), {min: 1, max: 1},
    pollutantKey + ' common strict 12/12 mask — all 6 years', false);
  Map.addLayer(originalSixOfSix.selfMask(), {min: 1, max: 1},
    pollutantKey + ' original-like 6/6 persistence', false);
  Map.addLayer(strictSixOfSix.selfMask(), {min: 1, max: 1},
    pollutantKey + ' strict 6/6 persistence', false);
  Map.addLayer(originalSixExcludedByStrict.selfMask(), {min: 1, max: 1},
    pollutantKey + ' 6/6 cells removed by strict completeness', false);
}

runPollutantAudit('NO2');
runPollutantAudit('CO');

print('==================================================');
print('AUDIT COMPLETE');
print('No source asset, publication result, or Master script was modified.');
